16  package org.mybatis.scripting.freemarker;
17  
18  import java.util.ArrayList;
19  import java.util.List;
20  
21  import org.junit.jupiter.api.Assertions;
22  import org.junit.jupiter.api.Test;
23  
24  import freemarker.template.Configuration;
25  import freemarker.template.TemplateModel;
26  import freemarker.template.TemplateModelException;
27  
28  /**
29   * Direct unit tests for {@link ParamObjectAdapter}. Note that {@link ParamObjectAdapter#getGeneratedParams()} has no
30   * caller in the FreeMarker template-processing pipeline (only {@link GeneratedParamsTemplateModel#getGeneratedParams()}
31   * is used by {@link MyBatisParamDirective}), so it cannot be exercised via a mapper/template integration test and is
32   * covered here directly.
33   */
34  class ParamObjectAdapterTest {
35  
36    @Test
37    void getGeneratedParamsReturnsSameListInstance() {
38      List<Object> generatedParams = new ArrayList<>();
39      ParamObjectAdapter adapter = new ParamObjectAdapter(new PreparedParam(), generatedParams,
40          Configuration.VERSION_2_3_22);
41  
42      Assertions.assertSame(generatedParams, adapter.getGeneratedParams());
43  
44      generatedParams.add("foo");
45      Assertions.assertEquals(1, adapter.getGeneratedParams().size());
46      Assertions.assertEquals("foo", adapter.getGeneratedParams().get(0));
47    }
48  
49    @Test
50    void isEmptyDelegatesToBeanModel() throws TemplateModelException {
51      ParamObjectAdapter adapter = new ParamObjectAdapter(new PreparedParam(), new ArrayList<>(),
52          Configuration.VERSION_2_3_22);
53  
54      // PreparedParam has readable bean properties, so the wrapped BeanModel is not empty.
55      Assertions.assertFalse(adapter.isEmpty());
56    }
57  
58    @Test
59    void getReturnsBeanProperty() throws TemplateModelException {
60      PreparedParam param = new PreparedParam();
61      ParamObjectAdapter adapter = new ParamObjectAdapter(param, new ArrayList<>(), Configuration.VERSION_2_3_22);
62  
63      Assertions.assertNotNull(adapter.get("innerObject"));
64    }
65  
66    @Test
67    void getFallsBackToAdditionalParams() throws TemplateModelException {
68      ParamObjectAdapter adapter = new ParamObjectAdapter(new PreparedParam(), new ArrayList<>(),
69          Configuration.VERSION_2_3_22);
70      TemplateModel additional = new MyBatisParamDirective();
71      adapter.putAdditionalParam("p", additional);
72  
73      Assertions.assertSame(additional, adapter.get("p"));
74    }
75  
76    @Test
77    void getReturnsGeneratedParamsWrapperForGeneratedParamsKey() throws TemplateModelException {
78      List<Object> generatedParams = new ArrayList<>();
79      ParamObjectAdapter adapter = new ParamObjectAdapter(new PreparedParam(), generatedParams,
80          Configuration.VERSION_2_3_22);
81  
82      TemplateModel model = adapter.get(FreeMarkerSqlSource.GENERATED_PARAMS_KEY);
83      Assertions.assertInstanceOf(GeneratedParamsTemplateModel.class, model);
84      Assertions.assertSame(generatedParams, ((GeneratedParamsTemplateModel) model).getGeneratedParams());
85    }
86  
87    @Test
88    void getReturnsNullForUnknownKey() throws TemplateModelException {
89      ParamObjectAdapter adapter = new ParamObjectAdapter(new PreparedParam(), new ArrayList<>(),
90          Configuration.VERSION_2_3_22);
91  
92      Assertions.assertNull(adapter.get("doesNotExist"));
93    }
94  }
